# Research agent — CONTAINER deployment, Python 3.11.
#
# The agent-server protocol adapter is preview-only, so it is pinned exactly:
# a floating range on a pre-release package is not reproducible. Stable
# dependencies use a lower bound plus a next-major upper bound so an upstream
# breaking change cannot silently reach production.
#
# Audit with `pip-audit` via scripts/check.ps1 before every deploy.

azure-ai-agentserver-responses==2.0.0b1
# Pinning the adapter alone does not pin the runtime: it declares
# `azure-ai-agentserver-core>=2.0.0b10`, an open-ended lower bound that pip
# resolves to prereleases. A rebuild silently picked up core 2.1.0b2, whose
# experimental resilient-task subsystem the hosted container environment does
# not provide, and every request failed with "Resilient task subsystem missing".
azure-ai-agentserver-core==2.0.0
azure-ai-projects>=2.4.0,<3.0.0
azure-identity>=1.25.3,<2.0.0
# Imported directly to build the API Management gateway client, so it is declared
# here rather than relied on as an azure-ai-projects transitive dependency.
openai>=1.109.1,<3.0.0
# Imported directly to record gen_ai token-usage spans. The hosted runtime
# configures the provider and exporter; only the API surface is used here.
opentelemetry-api>=1.44.0,<2.0.0
python-dotenv>=1.2.2,<2.0.0
